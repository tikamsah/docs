User Operation for maria db

CREATE USER 'user'@'%' IDENTIFIED BY 'your_password';

GRANT ALL PRIVILEGES ON *.* TO 'user'@'%' WITH GRANT OPTION;

GRANT ALL PRIVILEGES ON databas.tabele TO 'user'@'%';

GRANT SELECT, INSERT, UPDATE ON database_name.table_name TO 'username'@'host';

In /etc/mysql/mariadb.conf.d/50-server.cnf (Ubuntu/Debian) or /etc/my.cnf (CentOS):

[mysqld] bind-address = 0.0.0.0

mysql -u user -p -h host -P 3306 database_name

SELECT User, Host FROM mysql.user;

DROP USER 'username'@'host';

ALTER USER 'user'@'%' IDENTIFIED BY 'your_password';

FLUSH PRIVILEGES;

sudo systemctl restart mariadb
